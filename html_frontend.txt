const DIRS=[[0,0],[1,0],[-1,0],[0,1],[0,-1]];
function mulberry32(a){return function(){a|=0;a=a+0x6D2B79F5|0;let t=Math.imul(a^a>>>15,1|a);t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296}}

class Heap{
  constructor(){this.f=[];this.h=[];this.k=[]}
  get size(){return this.k.length}
  sw(i,j){const F=this.f,H=this.h,K=this.k;let t=F[i];F[i]=F[j];F[j]=t;t=H[i];H[i]=H[j];H[j]=t;t=K[i];K[i]=K[j];K[j]=t}
  push(f,h,k){
    const F=this.f,H=this.h,K=this.k;let i=K.length;F.push(f);H.push(h);K.push(k);
    while(i>0){const p=(i-1)>>1;if(F[p]<F[i]||(F[p]===F[i]&&H[p]<=H[i]))break;this.sw(i,p);i=p}
  }
  pop(){
    const F=this.f,H=this.h,K=this.k;const top=K[0];
    const lf=F.pop(),lh=H.pop(),lk=K.pop();const n=K.length;
    if(n>0){
      F[0]=lf;H[0]=lh;K[0]=lk;let i=0;
      for(;;){
        const l=2*i+1,r=l+1;let m=i;
        if(l<n&&(F[l]<F[m]||(F[l]===F[m]&&H[l]<H[m])))m=l;
        if(r<n&&(F[r]<F[m]||(F[r]===F[m]&&H[r]<H[m])))m=r;
        if(m===i)break;this.sw(i,m);i=m;
      }
    }
    return top;
  }
}
class Queue{constructor(){this.a=[];this.i=0}push(k){this.a.push(k)}pop(){return this.a[this.i++]}get size(){return this.a.length-this.i}}

function makeCtx(W,H,walls){
  const N=W*H,cache=new Map();
  const ctx={W,H,N,walls,maxT:160,maxNodes:250000,
    dist(g){
      if(cache.has(g))return cache.get(g);
      const d=new Int32Array(N).fill(-1);
      if(!walls[g]){
        const q=[g];d[g]=0;
        for(let i=0;i<q.length;i++){
          const c=q[i],x=c%W,y=(c/W)|0;
          for(let k=1;k<5;k++){
            const nx=x+DIRS[k][0],ny=y+DIRS[k][1];
            if(nx<0||ny<0||nx>=W||ny>=H)continue;
            const c2=ny*W+nx;if(walls[c2]||d[c2]>=0)continue;
            d[c2]=d[c]+1;q.push(c2);
          }
        }
      }
      cache.set(g,d);return d;
    }};
  return ctx;
}

/* Single-agent search on the plain grid (used by Independent mode). */
function searchGrid(ctx,s,g,algo){
  const {W,H,N,walls}=ctx;
  const par=new Int32Array(N).fill(-2),gc=new Int32Array(N),closed=new Uint8Array(N);
  const gx=g%W,gy=(g/W)|0;
  const hf=c=>Math.abs(c%W-gx)+Math.abs(((c/W)|0)-gy);
  const bfs=algo==='bfs';
  const open=bfs?new Queue():new Heap();
  par[s]=-1;gc[s]=0;
  if(bfs)open.push(s);else{const h=algo==='astar'?hf(s):0;open.push(h,h,s)}
  let nodes=0;
  while(open.size>0){
    const c=open.pop();
    if(closed[c])continue;closed[c]=1;nodes++;
    if(c===g){const p=[];for(let q=c;q!==-1;q=par[q])p.push(q);p.reverse();return{path:p,nodes}}
    const x=c%W,y=(c/W)|0;
    for(let k=1;k<5;k++){
      const nx=x+DIRS[k][0],ny=y+DIRS[k][1];
      if(nx<0||ny<0||nx>=W||ny>=H)continue;
      const c2=ny*W+nx;if(walls[c2]||closed[c2])continue;
      const ng=gc[c]+1;
      if(par[c2]===-2||(!bfs&&ng<gc[c2])){
        par[c2]=c;gc[c2]=ng;
        if(bfs)open.push(c2);else{const h=algo==='astar'?hf(c2):0;open.push(ng+h,h,c2)}
      }
    }
  }
  return{path:null,nodes};
}

/* Reservation table for prioritized planning. */
function newRes(N){return{vert:new Set(),edge:new Set(),lastT:new Int32Array(N).fill(-1),parkedAt:new Int32Array(N).fill(-1)}}
function addRes(res,N,p){
  for(let t=0;t<p.length;t++){
    res.vert.add(t*N+p[t]);
    if(t>res.lastT[p[t]])res.lastT[p[t]]=t;
    if(t+1<p.length&&p[t]!==p[t+1])res.edge.add((t*N+p[t])*N+p[t+1]);
  }
  res.parkedAt[p[p.length-1]]=p.length-1;
}
function parkRes(res,c){res.parkedAt[c]=0}

/* Space-time search: states are (cell, time); moves are 4 directions plus wait. */
function searchST(ctx,s,g,algo,res){
  const {W,H,N,walls,maxT}=ctx;
  const dg=ctx.dist(g);
  if(dg[s]<0)return{path:null,nodes:0};
  const hm=algo==='astar'?dg:null;
  const parent=new Int32Array((maxT+1)*N).fill(-2);
  parent[s]=-1;
  const bfs=algo==='bfs';
  const open=bfs?new Queue():new Heap();
  if(bfs)open.push(s);else{const h=hm?hm[s]:0;open.push(h,h,s)}
  let nodes=0;
  while(open.size>0){
    const key=open.pop();
    const t=(key/N)|0,c=key-t*N;
    nodes++;
    if(nodes>ctx.maxNodes)break;
    if(c===g&&t>res.lastT[g]&&res.parkedAt[g]===-1){
      const p=[];for(let q=key;q!==-1;q=parent[q])p.push(q%N);p.reverse();return{path:p,nodes};
    }
    if(t>=maxT)continue;
    const x=c%W,y=(c/W)|0,nt=t+1;
    for(let k=0;k<5;k++){
      const nx=x+DIRS[k][0],ny=y+DIRS[k][1];
      if(nx<0||ny<0||nx>=W||ny>=H)continue;
      const c2=ny*W+nx;
      if(walls[c2])continue;
      if(res.vert.has(nt*N+c2))continue;
      const pk=res.parkedAt[c2];if(pk!==-1&&nt>=pk)continue;
      if(c2!==c&&res.edge.has((t*N+c2)*N+c))continue;
      const k2=nt*N+c2;
      if(parent[k2]!==-2)continue;
      parent[k2]=key;
      if(bfs)open.push(k2);else{const h=hm?hm[c2]:0;open.push(nt+h,h,k2)}
    }
  }
  return{path:null,nodes};
}

/* ags: [{s,g}] cell indices. order: initial priority order. Every agent gets a path; unsolved agents hold at their start. */
function planAgents(ctx,ags,algo,mode,order0){
  const t0=performance.now(),n=ags.length,N=ctx.N;
  let nodes=0,reordered=false;
  const paths=new Array(n).fill(null);
  const order=order0&&order0.length===n?order0.slice():[...Array(n).keys()];
  let finalOrder=order,unsolved=new Set();
  if(mode==='independent'){
    for(let i=0;i<n;i++){
      const r=searchGrid(ctx,ags[i].s,ags[i].g,algo);nodes+=r.nodes;
      if(r.path)paths[i]=r.path;else{paths[i]=[ags[i].s];unsolved.add(i)}
    }
  }else{
    const dead=new Set();
    for(let i=0;i<n;i++)if(ctx.dist(ags[i].g)[ags[i].s]<0)dead.add(i);
    let active=order.filter(i=>!dead.has(i));
    let attempts=0,done=null,lastBad=-1;
    while(attempts<=n){
      const res=newRes(N);for(const i of dead)parkRes(res,ags[i].s);
      const tmp=new Array(n).fill(null);let bad=-1;
      for(const i of active){
        const r=searchST(ctx,ags[i].s,ags[i].g,algo,res);nodes+=r.nodes;
        if(!r.path){bad=i;break}
        tmp[i]=r.path;addRes(res,N,r.path);
      }
      if(bad<0){done=tmp;break}
      lastBad=bad;attempts++;
      if(active[0]===bad)break;
      active=[bad,...active.filter(i=>i!==bad)];reordered=true;
    }
    unsolved=new Set(dead);
    if(done){for(let i=0;i<n;i++)paths[i]=done[i]||[ags[i].s]}
    else{
      if(lastBad>=0)unsolved.add(lastBad);
      for(;;){
        const res=newRes(N);for(const i of unsolved)parkRes(res,ags[i].s);
        const tmp=new Array(n).fill(null);let bad=-1;
        for(const i of active){
          if(unsolved.has(i))continue;
          const r=searchST(ctx,ags[i].s,ags[i].g,algo,res);nodes+=r.nodes;
          if(!r.path){bad=i;break}
          tmp[i]=r.path;addRes(res,N,r.path);
        }
        if(bad<0){for(let i=0;i<n;i++)paths[i]=tmp[i]||[ags[i].s];break}
        unsolved.add(bad);
      }
    }
    finalOrder=active;
  }
  return{paths,failed:[...unsolved].sort((a,b)=>a-b),nodes,ms:performance.now()-t0,reordered,order:finalOrder};
}

/* Vertex conflicts (same cell, same tick) and swap conflicts (two agents trade cells). Agents rest at their last cell after arriving. */
function detectConflicts(paths){
  const out=[];let T=0;
  for(const p of paths)if(p)T=Math.max(T,p.length);
  const at=(p,t)=>p[Math.min(t,p.length-1)];
  for(let t=0;t<T;t++){
    const seen=new Map();
    for(let i=0;i<paths.length;i++){
      const p=paths[i];if(!p)continue;
      const c=at(p,t);
      if(seen.has(c)){for(const j of seen.get(c))out.push({type:'vertex',t,a:j,b:i,c1:c,c2:c});seen.get(c).push(i)}
      else seen.set(c,[i]);
    }
    if(t+1<T){
      for(let i=0;i<paths.length;i++){
        if(!paths[i])continue;
        const a0=at(paths[i],t),a1=at(paths[i],t+1);
        if(a0===a1)continue;
        for(let j=i+1;j<paths.length;j++){
          if(!paths[j])continue;
          if(at(paths[j],t)===a1&&at(paths[j],t+1)===a0)out.push({type:'swap',t,a:i,b:j,c1:a0,c2:a1});
        }
      }
    }
  }
  return out;
}

/* Scenario builders. Each returns {walls, agents:[{s,g}]}. */
function pickAgents(rng,walls,W,H,n,minD){
  const N=W*H,comp=new Int32Array(N).fill(-1);let best=[],id=0;
  for(let c=0;c<N;c++){
    if(walls[c]||comp[c]>=0)continue;
    const q=[c];comp[c]=id;
    for(let i=0;i<q.length;i++){
      const u=q[i],x=u%W,y=(u/W)|0;
      for(let k=1;k<5;k++){
        const nx=x+DIRS[k][0],ny=y+DIRS[k][1];
        if(nx<0||ny<0||nx>=W||ny>=H)continue;
        const v=ny*W+nx;if(walls[v]||comp[v]>=0)continue;comp[v]=id;q.push(v);
      }
    }
    if(q.length>best.length)best=q;id++;
  }
  const pool=best.slice();
  for(let i=pool.length-1;i>0;i--){const j=Math.floor(rng()*(i+1));const t=pool[i];pool[i]=pool[j];pool[j]=t}
  const m=Math.min(n,Math.floor(pool.length/2)),agents=[];
  const md=(a,b)=>Math.abs(a%W-b%W)+Math.abs(((a/W)|0)-((b/W)|0));
  for(let k=0;k<m;k++){
    const s=pool.shift();
    let gi=pool.findIndex(c=>md(s,c)>=minD);
    if(gi<0)gi=0;
    const g=pool.splice(gi,1)[0];
    agents.push({s,g});
  }
  return agents;
}
function genRandom(W,H,n,density,seed){
  const rng=mulberry32(seed),walls=new Uint8Array(W*H);
  for(let c=0;c<W*H;c++)walls[c]=rng()<density/100?1:0;
  return{walls,agents:pickAgents(rng,walls,W,H,n,8)};
}
function presetWarehouse(W,H,n,seed){
  const walls=new Uint8Array(W*H),rng=mulberry32(seed);
  for(const [y0,y1] of [[3,4],[7,8],[11,12]])for(const [x0,x1] of [[2,7],[10,15],[18,22]])
    for(let y=y0;y<=y1;y++)for(let x=x0;x<=x1;x++)walls[y*W+x]=1;
  return{walls,agents:pickAgents(rng,walls,W,H,n,10)};
}
function presetCorridor(W,H){
  const walls=new Uint8Array(W*H).fill(1),id=(x,y)=>y*W+x,my=8;
  for(let x=0;x<W;x++)walls[id(x,my)]=0;
  walls[id(11,my-1)]=0;
  return{walls,agents:[{s:id(0,my),g:id(W-1,my)},{s:id(W-1,my),g:id(0,my)}]};
}
function presetCrossing(W,H){
  const walls=new Uint8Array(W*H),id=(x,y)=>y*W+x;
  return{walls,agents:[
    {s:id(1,8),g:id(22,8)},{s:id(22,7),g:id(1,7)},
    {s:id(11,1),g:id(11,14)},{s:id(12,14),g:id(12,1)},
    {s:id(2,2),g:id(21,13)},{s:id(21,2),g:id(2,13)}]};
}

function replan(reason){
  const ctx=makeCtx(S.W,S.H,S.walls);
  const ags=S.agents.map(a=>({s:a.cur,g:a.goal}));
  const order=S.agents.map((_,i)=>i).sort((i,j)=>((S.agents[j].cur===S.agents[j].goal)-(S.agents[i].cur===S.agents[i].goal))||(i-j));
  const p=planAgents(ctx,ags,S.algo,S.mode,order);
  p.conflicts=detectConflicts(p.paths);
  p.cost=p.paths.map((x,i)=>p.failed.includes(i)?null:x.length-1);
  p.soc=p.cost.reduce((a,b)=>a+(b||0),0);
  p.makespan=p.cost.reduce((a,b)=>Math.max(a,b||0),0);
  p.heat=new Int32Array(S.W*S.H);p.heatMax=0;
  p.paths.forEach(x=>{new Set(x).forEach(c=>{p.heat[c]++;if(p.heat[c]>p.heatMax)p.heatMax=p.heat[c]})});
  S.plan=p;
  const nm=i=>'A'+S.agents[i].id;
  addLog(reason);
  addLog(ALG[S.algo]+' \u00B7 '+S.mode+' \u00B7 SOC '+p.soc+' \u00B7 makespan '+p.makespan+' \u00B7 '+p.conflicts.length+' conflict'+(p.conflicts.length===1?'':'s'));
  if(S.mode==='prioritized'&&p.reordered)addLog('Priority order changed to '+p.order.map(nm).join(' > ')+' to find a plan.');

  if(p.conflicts.length>5)addLog('  +'+(p.conflicts.length-5)+' more');
}

function editWall(c,val){
  if((S.walls[c]===1)===val)return false;
  if(val){
    if(S.agents.some(a=>a.goal===c||a.start===c||a.cur===c)){say('That cell is an agent start or goal.');return false}
  }
  const started=startedNow(),ti=Math.floor(S.tNow);
  if(started){
    const pos=S.agents.map((_,i)=>posAt(i,ti));
    if(val&&pos.includes(c)){say('An agent is standing on that cell right now.');return false}
    S.agents.forEach((a,i)=>{a.cur=pos[i]});
    S.elapsed+=ti;S.replans++;S.tNow=0;

    replan('Replan #'+S.replans+' at tick '+S.elapsed+': wall '+(val?'added':'removed')+' at ('+cx(c)+','+cy(c)+')');
  }else{

  }

}

    }
  }
}

  rows.forEach(r=>{h+='<tr><td>'+r.al+'</td><td>'+r.nodes.toLocaleString()+'</td><td>'+r.ms.toFixed(1)+'</td><td>'+r.soc+'</td><td>'+r.mk+'</td><td>'+r.conf+'</td></tr>'});
  $('#cmpWrap').innerHTML=h+'</tbody></table>';
  S.lastCmp='MAPF comparison | map: '+S.preset+' | agents: '+S.agents.length+' | mode: '+S.mode+(S.preset==='random'||S.preset==='warehouse'?' | seed: '+S.seed:'')+'\nAlgorithm\tNodes\tms\tSOC\tMakespan\tConflicts\n'+rows.map(r=>[r.al,r.nodes,r.ms.toFixed(1),r.soc,r.mk,r.conf].join('\t')).join('\n');
  $('#copyCmp').disabled=false;$('#cmpCopy').innerHTML='';
  addLog('Comparison run on '+S.agents.length+' agents ('+S.mode+').');
}
